// Program.cs — the records as JSON, the same SednaGraphData a call would send.
// JsonSerializerOptions.Web writes the camelCase names the script reads, whatever the
// app has done to its own JSON options.
app.MapGet("/api/service-map", async (IServiceCatalog catalog) =>
{
    var services = await catalog.ListAsync();
    return TypedResults.Json(new SednaGraphData(
        [.. services.Select(s => new SednaGraphNode(s.Id, s.Name) { Kind = s.Kind, Group = s.Team })],
        [.. services.SelectMany(s => s.Calls, (s, callee) => new SednaGraphEdge(s.Id, callee))]),
        JsonSerializerOptions.Web);
}).RequireAuthorization();

@* ServiceMap.razor — the browser fetches the records itself, with the page's cookies,
   and not one of them passes through the circuit. *@
@inject ISednaGraphs Graphs
@inject NavigationManager Nav

<div class="toolbar">
    <span class="toolbar-count" data-graph-for="service-map" data-graph-stats="{nodes} services"></span>
    <button class="btn btn--sm" type="button" @onclick="Refresh"><i class="ri-refresh-line"></i> <span>Refresh</span></button>
</div>
<div class="graph graph--viewport" data-graph data-graph-src="/api/service-map"
     data-graph-layout="islands" id="service-map" aria-label="Every service, by team"
     @onsedna-graph-open="Open" @onsedna-graph-open:preventDefault>
    <div class="graph-canvas"></div>
    <div class="graph-wait skeleton skeleton-block"></div>
    <div class="graph-empty graph-empty--error empty-state">
        <span class="empty-state-title">The service map could not be loaded</span>
    </div>
</div>

@code {
    // When the component learns the map changed — a deploy finished, a message on the app's
    // own hub — the browser fetches it again and shows what changed, as SetDataAsync does.
    private Task Refresh() => Graphs.ReloadAsync("service-map");

    private void Open(SednaGraphNodeEventArgs e) => Nav.NavigateTo($"/services/{e.Id}");
}
