# Coverage policy, read by build/coverage-gate.py. Rules: docs/coverage.md.
#
# THE RULE FOR BOTH NUMBERS: neither is ever raised because the suite exceeds it. Neither is a target.
# changed_lines_min is the share of the executable lines a pull request adds or changes that the
# suite must cover: it is how a change comes with tests. whole_floor is a fixed tripwire, set a
# little under what was measured, there to catch tests being deleted or a harness silently
# measuring less. A change that wants either number moved says why in its description, and the
# reviewer decides.
#
# Percentages, 0-100.
changed_lines_min = 90
whole_floor = 90

# What is measured: tracked files under a scope directory with one of its extensions.
scope = src/Sedna.UI/ .cs
scope = src/Sedna.UI.Catalogue/ .cs .razor
scope = src/Sedna.UI/js-parts/ .js
scope = src/Sedna.UI/graph-parts/ .js
scope = src/Sedna.UI/editor-parts/ .js
scope = src/Sedna.UI/chart-parts/ .js
scope = src/Sedna.UI/orb-parts/ .js
scope = src/Sedna.UI/signature-parts/ .js
scope = src/Sedna.UI/qr-parts/ .js

# A report that must be there. The gate fails when nothing was measured under one of these, so a
# job whose report went missing cannot raise the share by leaving its code out of the figure.
require = src/Sedna.UI/
require = src/Sedna.UI.Catalogue/
require = src/Sedna.UI/js-parts/
